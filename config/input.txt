################################################################

################################################################

################################################################
# Jumlah tahun lengkap minimum agar deret dianalisis (>= 4). 10 = templat hydrology_analysis.
min_years = 10
# Distribusi, metode momen (praktik SNI 2415:2016): Normal, LogNormal, Gumbel, LogPearson3
